 └── 📄 requirements.txt       import streamlit as st
import pandas as pd

st.set_page_config(layout="wide")
st.title("🏒 Interactive NHL Win Projection Model")

# 1. Sidebar for user inputs (Example: Adjusting Global Settings)
st.sidebar.header("Model Settings")
win_multiplier = st.sidebar.slider("Win/GAR Conversion Factor", 4.0, 8.0, 6.0, 0.5)
league_sh_avg = st.sidebar.slider("League Average Shooting %", 7.0, 10.0, 8.7, 0.1)

# 2. Hardcoded baseline data dictionary (Your exact code logic)
data = {
    'Team': ['Edmonton Oilers', 'Florida Panthers', 'Dallas Stars', 'Carolina Hurricanes', 'Chicago Blackhawks', 'San Jose Sharks'],
    'Baseline_xGF_60': [3.5, 3.4, 3.2, 3.6, 2.2, 2.0],
    'Baseline_xGA_60': [2.8, 2.6, 2.7, 2.5, 3.5, 3.7],
    'Net_Offseason_GAR': [12.5, -4.2, 2.1, -14.3, 18.2, 10.5],
    'Last_Season_SH_Pct': [10.2, 11.5, 9.8, 8.5, 7.2, 8.0],
    'Last_Season_SV_Pct': [90.0, 92.0, 91.3, 89.7, 90.3, 91.0],
    'New_Goalie_True_Talent_SV': [91.3, 92.2, 91.8, 91.3, 91.3, 89.5] 
}

df = pd.DataFrame(data)

# 3. Model Logic
df['Baseline_xG_Pct'] = df['Baseline_xGF_60'] / (df['Baseline_xGF_60'] + df['Baseline_xGA_60'])
df['Baseline_Wins'] = (df['Baseline_xG_Pct'] * 1.2 - 0.10) * 82
df['Offseason_Win_Adj'] = df['Net_Offseason_GAR'] / win_multiplier

df['SH_Deviation'] = df['Last_Season_SH_Pct'] - league_sh_avg
df['SH_Win_Adj'] = (df['SH_Deviation'] * -1) * 0.5 * 2.75

df['SV_Deviation_From_Talent'] = df['Last_Season_SV_Pct'] - df['New_Goalie_True_Talent_SV']
df['SV_Win_Adj'] = (df['SV_Deviation_From_Talent'] * -1) * 0.5 * 2.75

df['Goalie_Talent_Shift_Wins'] = (df['New_Goalie_True_Talent_SV'] - 91.3) * 2.75

df['Projected_Wins'] = (df['Baseline_Wins'] + df['Offseason_Win_Adj'] + df['SH_Win_Adj'] + df['SV_Win_Adj'] + df['Goalie_Talent_Shift_Wins']).round(1)
df['Total_PDO_Adj'] = (df['SH_Win_Adj'] + df['SV_Win_Adj'] + df['Goalie_Talent_Shift_Wins']).round(1)

# Format for output
df['Baseline_xG%'] = (df['Baseline_xG_Pct'] * 100).round(1).astype(str) + '%'
final_df = df[['Team', 'Baseline_xG%', 'New_Goalie_True_Talent_SV', 'Total_PDO_Adj', 'Projected_Wins']].sort_values(by='Projected_Wins', ascending=False)

# 4. Display interactive table on the website
st.dataframe(final_df, use_container_width=True)
